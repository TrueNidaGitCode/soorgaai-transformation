5. Eame — build the AI application

Illustrative

Eame assembles the AI application from Aria's structured data and Arth's infrastructure, and delivers the source code. 

// defect-matching-agent.js (illustrative)
import { hybridRetrieve } from "./retrieval.js";
import { generate } from "./llm.js";

export async function matchDefect({ description }) {
  const matches = await hybridRetrieve({
    queryText: description,
    sourceType: "defect",
    topK: 5,
  });

  const suggestion = await generate({
    systemPrompt: ROOT_CAUSE_PROMPT,
    userMessage: buildContext(description, matches),
  });

  return { matches, suggestedRootCause: suggestion.text };
}
